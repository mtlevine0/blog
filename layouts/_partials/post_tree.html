{{- /* Site-wide post tree, rendered on every page from baseof.html. Shown in the left margin on wide screens and as a collapsed "All posts" toggle above the content on narrow ones (see post-tree in custom.css and extend_footer.html). Series are folders of page bundles under content/posts (e.g. posts/video-wall/part-1-driver/). They aren't Hugo sections (no _index.md, so /posts/ keeps listing every part), so posts are grouped by the folder their bundle sits in. Titles of the form "Series, Part N: Subtitle" are split into a series heading and "Part N: Subtitle" entries. A `series` front matter param on the oldest post overrides the heading. A post can also be listed under other groups with a `groups` front matter list of folder names (e.g. groups: ["pcb-design"]). A name with no folder behind it makes a group of its own, headed by the humanized name. Words in $acronyms below are kept in capitals in any heading. Groups are sorted by heading, numbers before letters (A-Z, ignoring case). Posts in no group are collected into an "Other posts" group, which always comes last. */ -}} {{- $current := . }} {{- $posts := where site.RegularPages "Section" "posts" }} {{- $acronyms := slice "PCB" }} {{- $keys := slice }} {{- $ungrouped := slice }} {{- range $posts.ByDate.Reverse }} {{- $dir := strings.TrimSuffix "/" .File.Dir }} {{- if eq .BundleType "leaf" }}{{ $dir = path.Dir $dir }}{{ end }} {{- $own := slice }} {{- if ne $dir "posts" }}{{ $own = $own | append (path.Base $dir) }}{{ end }} {{- with .Params.groups }}{{ $own = $own | append . }}{{ end }} {{- if $own }}{{ $keys = $keys | append $own }}{{ else }}{{ $ungrouped = $ungrouped | append . }}{{ end }} {{- end }} {{- $groups := slice }} {{- range uniq $keys }} {{- $native := (where $posts "File.Dir" "like" (printf "^posts/%s/" .)).ByDate }} {{- $members := (union $native (where $posts "Params.groups" "intersect" (slice .))).ByDate }} {{- $series := . | humanize | title }} {{- with $native }} {{- $first := index . 0 }} {{- $series = path.Base (path.Dir (strings.TrimSuffix "/" $first.File.Dir)) | humanize | title }} {{- $split := index (split $first.LinkTitle ", Part ") 0 }} {{- if ne $split $first.LinkTitle }}{{ $series = $split }}{{ end }} {{- with $first.Params.series }}{{ $series = . }}{{ end }} {{- end }} {{- range $acronyms }}{{ $series = replaceRE (printf `(?i)\b%s\b` .) . $series }}{{ end }} {{- $groups = $groups | append (dict "heading" $series "sort" (lower $series) "members" $members "prefix" (printf "%s, " $series)) }} {{- end }} {{- $groups = sort $groups "sort" }} {{- with $ungrouped }} {{- $groups = $groups | append (dict "heading" "Other posts" "members" . "prefix" "") }} {{- end }}
